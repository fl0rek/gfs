import ue                                  # runtime API

def render():
    m   = ue.machine()                     # /machine provenance
    p   = ue.proc()                        # synthesized /proc
    out = [f"UNIVERSAL EXPLORER · {p.harness} · world up {p.uptime}"]
    out += section("machine", m.summary())
    out += bar("memory", p.mem.used, p.mem.limit)
    out += bar("cache",  p.cache.used, p.cache.limit)
    out += section("mounts", p.mounts)
    out += tree("processes", p.processes)
    return html(out)
